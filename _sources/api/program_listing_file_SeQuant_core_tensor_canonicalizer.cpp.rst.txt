   template <typename T>
   using get_support = decltype(std::get<0>(std::declval<T>()));
   
   template <typename T>
   constexpr bool is_tuple_like_v = meta::is_detected_v<get_support, T>;
   
   struct TensorBlockIndexComparer {
     TensorBlockIndexComparer() = default;
   
     explicit TensorBlockIndexComparer(container::set<Index> externals)
         : externals_(std::move(externals)) {}
   
     template <typename T>
     bool operator()(const T& lhs, const T& rhs) const {
       return compare<T>(lhs, rhs) < 0;
     }
   
     template <typename T>
     int compare(const T& lhs, const T& rhs) const {
       if constexpr (is_tuple_like_v<T>) {
         static_assert(
             std::tuple_size_v<T> == 2,
             "TensorBlockIndexComparer can only deal with tuple-like objects "
             "of size 2");
         const auto& lhs_first = std::get<0>(lhs);
         const auto& lhs_second = std::get<1>(lhs);
         const auto& rhs_first = std::get<0>(rhs);
         const auto& rhs_second = std::get<1>(rhs);
   
         static_assert(std::is_same_v<std::decay_t<decltype(lhs_first)>, Index>,
                       "TensorBlockIndexComparer can only work with indices");
         static_assert(std::is_same_v<std::decay_t<decltype(lhs_second)>, Index>,
                       "TensorBlockIndexComparer can only work with indices");
         static_assert(std::is_same_v<std::decay_t<decltype(rhs_first)>, Index>,
                       "TensorBlockIndexComparer can only work with indices");
         static_assert(std::is_same_v<std::decay_t<decltype(rhs_second)>, Index>,
                       "TensorBlockIndexComparer can only work with indices");
   
         // First compare only index spaces of equivalent pairs
         int res = compare_spaces(lhs_first, rhs_first);
         if (res != 0) {
           return res;
         }
   
         res = compare_spaces(lhs_second, rhs_second);
         if (res != 0) {
           return res;
         }
   
         // Then consider externality of equivalent pairs
         res = compare_externality(lhs_first, rhs_first);
         if (res != 0) {
           return res;
         }
   
         res = compare_externality(lhs_second, rhs_second);
         return res;
       } else {
         static_assert(std::is_same_v<std::decay_t<T>, Index>,
                       "TensorBlockIndexComparer can only work with indices");
   
         int res = compare_spaces(lhs, rhs);
         if (res != 0) {
           return res;
         }
   
         res = compare_externality(lhs, rhs);
         return res;
       }
     }
   
     int compare_spaces(const Index& lhs, const Index& rhs) const {
       if (lhs.basis() != rhs.basis()) {
         return lhs.basis() < rhs.basis() ? -1 : 1;
       }
   
       if (lhs.has_proto_indices() != rhs.has_proto_indices()) {
         return lhs.has_proto_indices() ? -1 : 1;
       }
   
       if (lhs.proto_indices().size() != rhs.proto_indices().size()) {
         return lhs.proto_indices().size() < rhs.proto_indices().size() ? -1 : 1;
       }
   
       for (std::size_t i = 0; i < lhs.proto_indices().size(); ++i) {
         const auto& lhs_proto = lhs.proto_indices()[i];
         const auto& rhs_proto = rhs.proto_indices()[i];
   
         int res = compare_spaces(lhs_proto, rhs_proto);
         if (res != 0) {
           return res;
         }
       }
   
       // Index spaces are equal
       return 0;
     }
   
     int compare_externality(const Index& lhs, const Index& rhs) const {
       if (externals_.empty()) return 0;
   
       const bool lhs_is_ext = externals_.find(lhs) != externals_.end();
       const bool rhs_is_ext = externals_.find(rhs) != externals_.end();
   
       if (lhs_is_ext != rhs_is_ext) {
         return lhs_is_ext ? -1 : 1;
       }
   
       return 0;
     }
   
    private:
     container::set<Index> externals_;
   };
   
   struct IndexTieBreakLess {
     template <typename T>
     bool operator()(const T& lhs, const T& rhs) const {
       if constexpr (is_tuple_like_v<T>) {
         static_assert(std::tuple_size_v<T> == 2,
                       "IndexTieBreakLess can only deal with tuple-like objects "
                       "of size 2");
   
         const Index& lhs_first = std::get<0>(lhs);
         const Index& lhs_second = std::get<1>(lhs);
         const Index& rhs_first = std::get<0>(rhs);
         const Index& rhs_second = std::get<1>(rhs);
   
         if (lhs_first != rhs_first) {
           return lhs_first < rhs_first;
         }
   
         return lhs_second < rhs_second;
       } else {
         return lhs < rhs;
       }
     }
   };
   
   struct TensorIndexComparer {
     template <typename T>
     bool operator()(const T& lhs, const T& rhs) const {
       const int res = TensorBlockIndexComparer{}.compare<T>(lhs, rhs);
       return res != 0 ? res < 0 : IndexTieBreakLess{}(lhs, rhs);
     }
   };
   
   template <typename Fallback>
   auto block_cmp_first(const TensorBlockIndexComparer& block_cmp,
                        Fallback fallback) {
     return [&block_cmp, fallback = std::move(fallback)](const auto& lhs,
                                                         const auto& rhs) {
       const int res = block_cmp.compare(lhs, rhs);
       return res != 0 ? res < 0 : fallback(lhs, rhs);
     };
   }

   container::set<Index> DefaultTensorCanonicalizer::external_indices_of(
       const AbstractTensor& t) const {
     container::set<Index> result;
     if (external_indices_.empty()) return result;
     for (const Index& idx : slots(t)) {
       if (external_indices_.find(idx) != external_indices_.end())
         result.insert(idx);
     }
     return result;
   }
   
   void DefaultTensorCanonicalizer::canonicalize_braket(AbstractTensor& t) {
     if (t._braket_symmetry() != BraKetSymmetry::Symm) {
       return;
     }
   
     // bra<->ket exchange is a symmetry for braket-symmetric tensors, so pick a
     // canonical orientation. The choice is governed solely by the canonical
     // "colors" of the bra and ket bundles -- i.e. their index spaces, not the
     // index labels -- so the result is label-independent. Bundles with identical
     // spaces (e.g. g{p,q;r,s}) compare equal and are left untouched; only
     // differing-color bundles are reoriented (so e.g. a half-tensor X{;a;x} folds
     // into X{a;;x}).
     const TensorBlockIndexComparer cmp;
     auto space_less = [&cmp](const Index& a, const Index& b) {
       return cmp.compare_spaces(a, b) < 0;
     };
   
     auto bra = mutable_bra_range(t);
     auto ket = mutable_ket_range(t);
   
     // Compare the bundles by their space sequences *sorted by color*, so the
     // decision is independent of the within-bundle index order. Column/perm
     // symmetry can permute the bra (and ket) order without changing the tensor,
     // and a comparison over the as-given order could otherwise pick different
     // orientations for equivalent inputs.
     std::vector<Index> bra_spaces, ket_spaces;
     for (auto&& idx : bra) bra_spaces.push_back(idx);
     for (auto&& idx : ket) ket_spaces.push_back(idx);
   
     ranges::sort(bra_spaces, space_less);
     ranges::sort(ket_spaces, space_less);
   
     // canonical orientation: the bundle whose spaces are lexicographically
     // larger goes to bra.
     if (ranges::lexicographical_compare(bra_spaces, ket_spaces, space_less)) {
       t._swap_bra_ket();
     }
   }
   
   ExprPtr DefaultTensorCanonicalizer::apply(AbstractTensor& t) const {
     canonicalize_braket(t);
   
     const auto ctx = get_default_context_snapshot();
     const auto& idxcmp = ctx.index_comparer();
     const auto& paircmp = ctx.index_pair_comparer();
   
     // indices are ordered by space, then external indices go first, then the
     // context's comparers order the rest; the default comparers start with the
     // space comparison, so only their tie-break is left to do
     const bool default_idxcmp =
         idxcmp.target_type() == typeid(TensorIndexComparer);
     const bool default_paircmp =
         paircmp.target_type() == typeid(TensorIndexComparer);
     auto idx_fallback = [&](const auto& lhs, const auto& rhs) {
       return default_idxcmp ? IndexTieBreakLess{}(lhs, rhs) : idxcmp(lhs, rhs);
     };
     auto pair_fallback = [&](const auto& lhs, const auto& rhs) {
       return default_paircmp ? IndexTieBreakLess{}(lhs, rhs) : paircmp(lhs, rhs);
     };
   
     const TensorBlockIndexComparer block_cmp(external_indices_of(t));
     return this->apply(t, block_cmp_first(block_cmp, idx_fallback),
                        block_cmp_first(block_cmp, pair_fallback));
   }
   
   ExprPtr TensorBlockCanonicalizer::apply(AbstractTensor& t) const {
     canonicalize_braket(t);
   
     const TensorBlockIndexComparer block_cmp(external_indices_of(t));
     return DefaultTensorCanonicalizer::apply(t, block_cmp, block_cmp);
   }
